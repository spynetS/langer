The code is not that neat because I focused on getting a working
compiler done instead of good code.

}

#+end_src

** Ptr example

#+begin_src c
extern puts(string fmt): void;
extern printf(string fmt, int a): void;
extern malloc(int a): int;

func set (int ptr) {
  let i :int = 0;
  while i < 10 {
      ptr[i] = i+1;
      i = i + 1;
  }
}


func main() : int {
  puts("Hello World!")
  let age := malloc(10*8);
  set(age)
  let i := 0;
  while i < 10 {
      printf("%d ", age[i])
      i = i + 1;
  }

  return 0;
}
#+end_src
